using System;

class Program
{
    static void Main()
    {
        int x = 1; // Начальное значение аргумента

        while (x <= 5) // Цикл выполняется, пока x от 1 до 5 включительно
        {
            // Вычисляем x^2 с помощью Math.Pow()
            double xSquared = Math.Pow(x, 2);
            
            // Вычисляем значение функции y = x^2 + 3x + 2
            double y = xSquared + 3 * x + 2;

            // Выводим аргумент и значение функции на отдельной строке
            Console.WriteLine($"x = {x}, y = {y}");

            // Изменяем значение x для перехода к следующей итерации
            x++; 
        }
    }
}
